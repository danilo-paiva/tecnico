# Endpoints — Gestão de Eventos & Ingressos (PAW 3BI com JWT)
# Base: http://127.0.0.1:8001
# Front: http://127.0.0.1:8001/login.html (ana@email.com / Senha@123)
# Todas as rotas de recursos exigem: Authorization: Bearer <token>

# ========== AUTH (público) ==========
POST http://127.0.0.1:8001/auth/login
Content-Type: application/json
{
  "email": "ana@email.com",
  "senha": "Senha@123"
}
# resposta: {success:true, data:{token:"eyJ...", expiresIn:7200, usuario:{idParticipante, nome, email}}}

POST http://127.0.0.1:8001/auth/login

# ========== cURL com JWT (copiar token do login) ==========
# Login e salvar token
TOKEN=$(curl -s -X POST http://127.0.0.1:8001/auth/login -H "Content-Type: application/json" -d '{"email":"ana@email.com","senha":"Senha@123"}' | php -r '$j=json_decode(file_get_contents("php://stdin"),true); echo $j["data"]["token"];')
# Listar com token
curl -s http://127.0.0.1:8001/locais -H "Authorization: Bearer $TOKEN" | jq
curl -s http://127.0.0.1:8001/eventos -H "Authorization: Bearer $TOKEN" | jq
# Criar com token
curl -s -X POST http://127.0.0.1:8001/locais -H "Content-Type: application/json" -H "Authorization: Bearer $TOKEN" -d '{"local":{"nome":"Local Via cURL JWT","endereco":"Rua cURL 99","capacidade":200}}' | jq
# Front: abrir http://127.0.0.1:8001/login.html e logar
